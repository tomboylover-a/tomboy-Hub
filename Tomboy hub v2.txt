--------------------------------------------------
-- 1. EVENT-DRIVEN ATTACK SPEED (OPTIMIZED)
--------------------------------------------------
local function applyBuffToTool(tool)
    if not tool:IsA("Tool") then return end
    
    local buff = tool:FindFirstChild("SwingSpeedBuff")
    if L.attackSpeedEnabled then
        if not buff then
            buff = Instance_new("NumberValue")
            buff.Name = "SwingSpeedBuff"
            buff.Parent = tool
        end
        buff.Value = L.attackSpeedMultiplier
    else
        if buff then buff:Destroy() end
    end
end

function L.updateAllTools()
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    
    if char then
        for _, tool in char:GetChildren() do applyBuffToTool(tool) end
    end
    if backpack then
        for _, tool in backpack:GetChildren() do applyBuffToTool(tool) end
    end
end

-- Bind to character and backpack to catch equips/unequips automatically
local function setupToolListeners(char)
    char.ChildAdded:Connect(applyBuffToTool)
    local backpack = LocalPlayer:WaitForChild("Backpack", 5)
    if backpack then
        backpack.ChildAdded:Connect(applyBuffToTool)
    end
end

if LocalPlayer.Character then setupToolListeners(LocalPlayer.Character) end

--------------------------------------------------
-- 2. HITBOX LOGIC (OPTIMIZED)
--------------------------------------------------
local function addHitboxesToZombie(zombie)
    if not L.zombieHitboxEnabled or L.zombieHitboxAddedParts[zombie] then return end
    local hrp = zombie:FindFirstChild("HumanoidRootPart")
    local head = zombie:FindFirstChild("Head")
    if not hrp or not head then return end

    local targetTransparency = L.zombieHitboxVisible and L.zombieHitboxTransparency or 1
    local hitboxColor = Color3_fromRGB(160, 32, 240)

    local outer = Instance_new("Part")
    outer.Name = "ZombieHitbox_Outer"
    outer.Size = Vector3_new(L.zombieHitboxSize, L.zombieHitboxSize, L.zombieHitboxSize)
    outer.Color = hitboxColor
    outer.Material = Enum.Material.Neon
    outer.Transparency = targetTransparency
    outer.CanCollide = false
    outer.CanTouch = true
    outer.Massless = true
    outer.Anchored = false
    outer.CFrame = hrp.CFrame
    outer.Parent = zombie

    local weldOuter = Instance_new("WeldConstraint")
    weldOuter.Part0 = hrp
    weldOuter.Part1 = outer
    weldOuter.Parent = outer

    local headBox = Instance_new("Part")
    headBox.Name = "ZombieHitbox_Head"
    headBox.Size = Vector3_new(L.zombieHitboxSize/2, L.zombieHitboxSize/2, L.zombieHitboxSize/2)
    headBox.Color = hitboxColor
    headBox.Material = Enum.Material.Neon
    headBox.Transparency = targetTransparency
    headBox.CanCollide = false
    headBox.CanTouch = true
    headBox.Massless = true
    headBox.Anchored = false
    headBox.CFrame = head.CFrame
    headBox.Parent = zombie

    local weldHead = Instance_new("WeldConstraint")
    weldHead.Part0 = head
    weldHead.Part1 = headBox
    weldHead.Parent = headBox

    L.zombieHitboxAddedParts[zombie] = { outer = outer, head = headBox }
end

local function removeHitboxesFromZombie(zombie)
    local parts = L.zombieHitboxAddedParts[zombie]
    if parts then
        if parts.outer then parts.outer:Destroy() end
        if parts.head then parts.head:Destroy() end
        L.zombieHitboxAddedParts[zombie] = nil
    end
end

local function refreshZombieHitboxes()
    if not L.zombieHitboxEnabled then
        for zombie in L.zombieHitboxAddedParts do
            removeHitboxesFromZombie(zombie)
        end
        L.zombieHitboxAddedParts = {}
        return
    end

    local folders = {workspace:FindFirstChild("Zombies"), workspace:FindFirstChild("Camera")}
    for _, folder in folders do
        if folder then
            for _, z in folder:GetChildren() do
                if z:IsA("Model") and (z.Name == "m_Zombie" or z:FindFirstChild("Orig")) then
                    addHitboxesToZombie(z)
                end
            end
        end
    end
end

local function updateAllZombieHitboxSizes()
    if not L.zombieHitboxEnabled then return end
    for _, parts in L.zombieHitboxAddedParts do
        if parts.outer and parts.outer.Parent then
            parts.outer.Size = Vector3_new(L.zombieHitboxSize, L.zombieHitboxSize, L.zombieHitboxSize)
        end
        if parts.head and parts.head.Parent then
            parts.head.Size = Vector3_new(L.zombieHitboxSize/2, L.zombieHitboxSize/2, L.zombieHitboxSize/2)
        end
    end
end

local function updateHitboxVisibility()
    local targetTransparency = L.zombieHitboxVisible and L.zombieHitboxTransparency or 1
    for _, parts in L.zombieHitboxAddedParts do
        if parts.outer and parts.outer.Parent then
            parts.outer.Transparency = targetTransparency
        end
        if parts.head and parts.head.Parent then
            parts.head.Transparency = targetTransparency
        end
    end
end

local function onZombieAdded(zombie)
    if L.zombieHitboxEnabled and zombie:IsA("Model") and zombie.Name == "m_Zombie" then
        task_wait(0.1)
        addHitboxesToZombie(zombie)
    end
end

local zombiesFolder = workspace:FindFirstChild("Zombies")
if zombiesFolder then zombiesFolder.ChildAdded:Connect(onZombieAdded) end
local cameraFolder = workspace:FindFirstChild("Camera")
if cameraFolder then cameraFolder.ChildAdded:Connect(onZombieAdded) end

-- Optimized background cleanup loop
task.spawn(function()
    while true do
        task_wait(1)
        if L.zombieHitboxEnabled then
            for zombie in L.zombieHitboxAddedParts do
                if not zombie:IsDescendantOf(workspace) or not zombie:FindFirstChild("HumanoidRootPart") then
                    removeHitboxesFromZombie(zombie)
                end
            end
        end
    end
end)

--------------------------------------------------
-- 3. COMBAT HOOKS (OPTIMIZED)
--------------------------------------------------
local function getZombieModel(part)
    if not part then return nil end
    local model = part:FindFirstAncestor("m_Zombie")
    if model then return model end
    
    local current = part.Parent
    for i = 1, 3 do
        if not current or current == workspace then break end
        if current:FindFirstChild("Orig") then return current end
        current = current.Parent
    end
    return nil
end

local function getZombieHead(zombieModel)
    return zombieModel:FindFirstChild("Head") or zombieModel:FindFirstChild("Head", true)
end

function L.unifiedBayonetHitCheck(self, origin, direction, raycastParams, hitEntities)
    local rayResult = workspace:Raycast(origin, direction, raycastParams)
    if rayResult then
        local hitPart = rayResult.Instance
        local zombieModel = getZombieModel(hitPart)
        if zombieModel then
            local now = tick()
            if L.lastZombieHitTime[zombieModel] and now - L.lastZombieHitTime[zombieModel] < L.ZOMBIE_HIT_COOLDOWN then
                return 0
            end
            
            local orig = zombieModel:FindFirstChild("Orig")
            local head = getZombieHead(zombieModel)
            
            if orig and head then
                local zombieRef = orig.Value
                local hitPos = head.Position
                local partName = (L.headshotEnabled or hitPart == head) and "Head" or "Torso"

                if L.removeBloodEnabled then
                    local rootPart = zombieModel:FindFirstChild("HumanoidRootPart") or zombieModel:FindFirstChild("Torso")
                    if rootPart then hitPos = rootPart.Position + Vector3_new(0, -9999, 0) end
                end
                
                local char = LocalPlayer.Character
                local headPart = char and char:FindFirstChild("Head")
                local Direction = headPart and (hitPos - headPart.Position).Unit or Vector3_new(0, 1, 0)
                
                self.remoteEvent:FireServer("Bayonet_HitZombie", zombieRef, hitPos, true, partName)
                self.remoteEvent:FireServer("HitZombieM", zombieRef, hitPos, true, hitPos, partName, Direction)
                
                L.lastZombieHitTime[zombieModel] = now
                return 1
            end
        end
    end
    
    if L.originalBayonetHitCheck then
        return L.originalBayonetHitCheck(self, origin, direction, raycastParams, hitEntities)
    end
    return 0
end

function L.unifiedMeleeHitCheck(self, origin, direction, raycastParams, hitEntities, isCharge)
    local rayResult = workspace:Raycast(origin, direction, raycastParams)
    if rayResult then
        local hitPart = rayResult.Instance
        local zombieModel = getZombieModel(hitPart)
        if zombieModel then
            local now = tick()
            if L.lastZombieHitTime[zombieModel] and now - L.lastZombieHitTime[zombieModel] < L.ZOMBIE_HIT_COOLDOWN then
                return 0
            end
            
            local orig = zombieModel:FindFirstChild("Orig")
            local head = getZombieHead(zombieModel)
            
            if orig and head then
                local zombieRef = orig.Value
                local hitPos = head.Position
                local partName = (L.headshotEnabled or hitPart == head) and "Head" or "Torso"

                if L.removeBloodEnabled then
                    local rootPart = zombieModel:FindFirstChild("HumanoidRootPart") or zombieModel:FindFirstChild("Torso")
                    if rootPart then hitPos = rootPart.Position + Vector3_new(0, -9999, 0) end
                end
                
                local char = LocalPlayer.Character
                local headPart = char and char:FindFirstChild("Head")
                local Direction = headPart and (hitPos - headPart.Position).Unit or Vector3_new(0, 1, 0)
                
                if isCharge then
                    self.remoteEvent:FireServer("ThrustCharge", zombieRef, hitPos, rayResult.Normal)
                else
                    self.remoteEvent:FireServer("HitZombieM", zombieRef, hitPos, true, hitPos, partName, Direction)
                end
                
                L.lastZombieHitTime[zombieModel] = now
                return 1
            end
        end
    end
    
    if L.originalMeleeHitCheck then
        return L.originalMeleeHitCheck(self, origin, direction, raycastParams, hitEntities, isCharge)
    end
    return 0
end

function L.updateHitHooks()
    local shouldHook = L.headshotEnabled or L.removeBloodEnabled
    local rs = ReplicatedStorage
    local weapons = rs:FindFirstChild("Modules") and rs.Modules:FindFirstChild("Weapons")
    local canHookFunction = type(hookfunction) == "function"
    
    if shouldHook and not L.hookInstalled then
        if weapons then
            local success, FlintLock = pcall(require, weapons:FindFirstChild("Flintlock"))
            if success and FlintLock then
                if canHookFunction then
                    L.originalBayonetHitCheck = hookfunction(FlintLock.BayonetHitCheck, L.unifiedBayonetHitCheck)
                else
                    L.originalBayonetHitCheck = L.originalBayonetHitCheck or FlintLock.BayonetHitCheck
                    FlintLock.BayonetHitCheck = L.unifiedBayonetHitCheck
                end
            end
            local success2, MeleeBase = pcall(require, weapons:FindFirstChild("MeleeBase"))
            if success2 and MeleeBase then
                if canHookFunction then
                    L.originalMeleeHitCheck = hookfunction(MeleeBase.MeleeHitCheck, L.unifiedMeleeHitCheck)
                else
                    L.originalMeleeHitCheck = L.originalMeleeHitCheck or MeleeBase.MeleeHitCheck
                    MeleeBase.MeleeHitCheck = L.unifiedMeleeHitCheck
                end
            end
        end
        L.hookInstalled = true
    elseif not shouldHook and L.hookInstalled then
        if weapons then
            if L.originalBayonetHitCheck then
                local success, FlintLock = pcall(require, weapons:FindFirstChild("Flintlock"))
                if success and FlintLock then
                    if canHookFunction then hookfunction(FlintLock.BayonetHitCheck, L.originalBayonetHitCheck) else FlintLock.BayonetHitCheck = L.originalBayonetHitCheck end
                end
            end
            if L.originalMeleeHitCheck then
                local success, MeleeBase = pcall(require, weapons:FindFirstChild("MeleeBase"))
                if success and MeleeBase then
                    if canHookFunction then hookfunction(MeleeBase.MeleeHitCheck, L.originalMeleeHitCheck) else MeleeBase.MeleeHitCheck = L.originalMeleeHitCheck end
                end
            end
        end
        L.hookInstalled = false
    end
end

LocalPlayer.CharacterAdded:Connect(function(char)
    setupToolListeners(char)
    task_wait(1)
    L.updateHitHooks()
    L.updateAllTools()
end)
